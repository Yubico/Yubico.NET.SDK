cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS deployment target")
endif()
project(hidinput_harness C)
if(NOT APPLE)
    message(FATAL_ERROR "macOS-only regression harness")
endif()
set(CMAKE_C_STANDARD 11)
include(../cmake/hidinput-common.cmake)
add_library(hidinput_harness SHARED ${HIDINPUT_COMMON_SOURCES} backend_synthetic.c)
target_compile_options(hidinput_harness PRIVATE -Wall -Wextra -Werror ${HIDINPUT_COMPILE_OPTIONS})
target_link_libraries(hidinput_harness PRIVATE ${HIDINPUT_FRAMEWORKS})
add_executable(hidinput_tests tests.c)
target_link_libraries(hidinput_tests PRIVATE hidinput_harness)
enable_testing()
foreach(n RANGE 1 23)
    add_test(NAME hidinput_${n} COMMAND hidinput_tests ${n})
    set_tests_properties(hidinput_${n} PROPERTIES TIMEOUT 5)
endforeach()
