cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS deployment target")
endif()
project(hidinput_experiment C)
if(NOT APPLE)
    message(FATAL_ERROR "macOS-only experiment")
endif()
set(CMAKE_C_STANDARD 11)
add_library(hidinput_experiment SHARED owner.c backend_synthetic.c backend_iohid.c)
target_compile_options(hidinput_experiment PRIVATE -Wall -Wextra -Werror -fblocks)
target_link_libraries(hidinput_experiment PRIVATE "-framework IOKit" "-framework CoreFoundation")
add_executable(hidinput_tests tests.c)
target_link_libraries(hidinput_tests PRIVATE hidinput_experiment)
enable_testing()
foreach(n RANGE 1 23)
    add_test(NAME hidinput_${n} COMMAND hidinput_tests ${n})
    set_tests_properties(hidinput_${n} PROPERTIES TIMEOUT 5)
endforeach()
